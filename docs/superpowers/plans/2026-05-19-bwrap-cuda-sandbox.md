# Bwrap CUDA Sandbox Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add explicit, permission-profile-scoped CUDA access to the Linux bubblewrap sandbox without changing default sandbox behavior.

**Architecture:** Extend the runtime `PermissionProfile::Managed` shape with default-empty hardware permissions, then parse `[permissions.<name>.hardware]` into that runtime model. The Linux sandbox helper receives the existing permission-profile JSON, derives CUDA access from it, appends compute-only NVIDIA `--dev-bind-try` mounts when enabled, and marks bwrap-launched commands with `SANDBOX_RUNTIME=bwrap` so runtime libraries can detect the sandbox.

**Tech Stack:** Rust, serde, schemars, ts-rs, bubblewrap, Cargo tests, Codex `just` recipes.

---

- [ ] **Step 4: Add compute-only CUDA bind helpers**

Add these helpers in `codex-rs/linux-sandbox/src/bwrap.rs` near `path_to_string()`:

```rust
fn append_cuda_device_bind_args(
    args: &mut Vec<String>,
    dev_root: &Path,
    hardware_permissions: HardwarePermissions,
) {
    if !hardware_permissions.cuda {
        return;
    }

    for (source, destination) in cuda_device_bind_mounts(dev_root) {
        args.push("--dev-bind-try".to_string());
        args.push(path_to_string(&source));
        args.push(path_to_string(&destination));
    }
}

fn cuda_device_bind_mounts(dev_root: &Path) -> Vec<(PathBuf, PathBuf)> {
    let mut mounts = Vec::new();
    for name in ["nvidiactl", "nvidia-uvm", "nvidia-uvm-tools", "nvidia-caps"] {
        mounts.push((
            dev_root.join(name),
            PathBuf::from("/dev").join(name),
        ));
    }

    mounts.extend(
        discover_numbered_nvidia_devices(dev_root)
            .into_iter()
            .map(|name| (dev_root.join(&name), PathBuf::from("/dev").join(name))),
    );
    mounts
}

fn discover_numbered_nvidia_devices(dev_root: &Path) -> Vec<String> {
    let Ok(entries) = fs::read_dir(dev_root) else {
        return Vec::new();
    };

    let mut devices = entries
        .filter_map(|entry| entry.ok())
        .filter_map(|entry| entry.file_name().into_string().ok())
        .filter_map(|name| {
            let index = name.strip_prefix("nvidia")?.parse::<u32>().ok()?;
            Some((index, name))
        })
        .collect::<Vec<_>>();
    devices.sort_by_key(|(index, _)| *index);
    devices.into_iter().map(|(_, name)| name).collect()
}
```

#![cfg(target_os = "linux")]
#![allow(clippy::unwrap_used)]

use codex_core::exec_env::create_env;
use codex_protocol::config_types::ShellEnvironmentPolicy;
use codex_protocol::models::HardwarePermissions;
use codex_protocol::models::PermissionProfile;
use std::collections::HashMap;
use std::path::Path;
use std::process::Output;
use std::process::Stdio;
use std::time::Duration;
use tokio::process::Command;

const CUDA_TIMEOUT_MS: u64 = 10_000;
const CUDA_HOST_INIT_PROBE: &str = r#"
import ctypes
from ctypes import byref, c_int

cuda = ctypes.CDLL("libcuda.so.1")
result = cuda.cuInit(0)
if result != 0:
    raise SystemExit(f"cuInit failed: {result}")
count = c_int()
result = cuda.cuDeviceGetCount(byref(count))
if result != 0:
    raise SystemExit(f"cuDeviceGetCount failed: {result}")
if count.value < 1:
    raise SystemExit("cuDeviceGetCount returned no devices")
print(f"CUDA devices: {count.value}")
"#;
const CUDA_SANDBOX_INIT_PROBE: &str = r#"
import ctypes
import os
from ctypes import byref, c_int

sandbox_runtime = os.environ.get("SANDBOX_RUNTIME")
if sandbox_runtime != "bwrap":
    raise SystemExit(f"unexpected SANDBOX_RUNTIME: {sandbox_runtime!r}")

cuda = ctypes.CDLL("libcuda.so.1")
result = cuda.cuInit(0)
if result != 0:
    raise SystemExit(f"cuInit failed: {result}")
count = c_int()
result = cuda.cuDeviceGetCount(byref(count))
if result != 0:
    raise SystemExit(f"cuDeviceGetCount failed: {result}")
if count.value < 1:
    raise SystemExit("cuDeviceGetCount returned no devices")
print(f"CUDA devices: {count.value}")
"#;

fn create_env_from_core_vars() -> HashMap<String, String> {
    let policy = ShellEnvironmentPolicy::default();
    create_env(&policy, /*thread_id*/ None)
}

async fn run_command(command: &[&str], permission_profile: &PermissionProfile) -> Output {
    let cwd = std::env::current_dir().expect("cwd should exist");
    let permission_profile_json =
        serde_json::to_string(permission_profile).expect("permission profile should serialize");
    let mut args = vec![
        "--sandbox-policy-cwd".to_string(),
        cwd.to_string_lossy().to_string(),
        "--permission-profile".to_string(),
        permission_profile_json,
        "--".to_string(),
    ];
    args.extend(command.iter().map(|entry| (*entry).to_string()));

    let mut cmd = Command::new(env!("CARGO_BIN_EXE_codex-linux-sandbox"));
    cmd.args(args)
        .current_dir(cwd)
        .env_clear()
        .envs(create_env_from_core_vars())
        .stdin(Stdio::null())
        .stdout(Stdio::piped())
        .stderr(Stdio::piped());

    match tokio::time::timeout(Duration::from_millis(CUDA_TIMEOUT_MS), cmd.output()).await {
        Ok(Ok(output)) => output,
        Ok(Err(err)) => panic!("sandbox command should execute: {err}"),
        Err(err) => panic!("sandbox command should not time out: {err}"),
    }
}

async fn host_cuda_probe_succeeds() -> bool {
    if !Path::new("/dev/nvidiactl").exists() {
        return false;
    }

    let mut cmd = Command::new("python3");
    cmd.arg("-c")
        .arg(CUDA_HOST_INIT_PROBE)
        .stdin(Stdio::null())
        .stdout(Stdio::piped())
        .stderr(Stdio::piped());

    match tokio::time::timeout(Duration::from_millis(CUDA_TIMEOUT_MS), cmd.output()).await {
        Ok(Ok(output)) => output.status.success(),
        Ok(Err(_)) | Err(_) => false,
    }
}

#[tokio::test]
async fn cuda_profile_allows_cuda_driver_init_under_bwrap() {
    if !host_cuda_probe_succeeds().await {
        eprintln!("skipping CUDA sandbox smoke test: host CUDA probe is unavailable");
        return;
    }

    let permission_profile =
        PermissionProfile::read_only().with_hardware_permissions(HardwarePermissions {
            cuda: true,
        });

    let output = run_command(
        &["python3", "-c", CUDA_SANDBOX_INIT_PROBE],
        &permission_profile,
    )
    .await;

    assert!(
        output.status.success(),
        "expected CUDA init and SANDBOX_RUNTIME=bwrap detection to succeed under bwrap; status={:?}; stdout={}; stderr={}",
        output.status.code(),
        String::from_utf8_lossy(&output.stdout),
        String::from_utf8_lossy(&output.stderr)
    );
}

```markdown
- Commands launched through the bubblewrap runtime receive
  `SANDBOX_RUNTIME=bwrap` via bubblewrap `--setenv`. This marker is only set
  when bwrap actually starts and is intended for runtime libraries, including
  CUDA-specific code paths, that need to detect the sandbox runtime.
- When the active permissions profile sets `hardware.cuda = true`, bubblewrap
  bind-mounts compute-only NVIDIA CUDA device nodes with `--dev-bind-try`.
  This includes `/dev/nvidiactl`, `/dev/nvidia[0-9]+`,
  `/dev/nvidia-uvm`, `/dev/nvidia-uvm-tools`, and `/dev/nvidia-caps`
  when present. Display-oriented nodes such as `/dev/nvidia-modeset` are not
  included.
```
